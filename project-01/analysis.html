#Looking at average sleep distribution to ensure it looks normal
ggplot(sleep,aes(x=AverageSleep)) +
  geom_histogram(binwidth = .3,fill="blue",color="black") +
  labs(
    title="Average Sleep Distribution",
    x="Average Sleep (hours)",
    y="Number of Students"
  ) + theme_classic()

#Looking at avg sleep vs gpa
ggplot(sleep, aes(x = AverageSleep, y = GPA,color=factor(ClassYear))) +
  geom_point() +
  geom_smooth(se=FALSE) +
  labs(
    title = "Average Sleep vs GPA",
    color="Class Year",
    x = "Average Sleep (Hours)",
    y = "GPA"
  ) + theme_minimal()
## `geom_smooth()` using method = 'loess' and formula = 'y ~ x'
